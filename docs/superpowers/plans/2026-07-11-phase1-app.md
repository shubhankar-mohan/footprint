# Phase 1 App Implementation Plan (SwiftUI menu-bar, SPM)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax.

**Goal:** A native macOS menu-bar app (SwiftUI `MenuBarExtra`, built with SPM — no Xcode) that supervises the Node bridge, renders every Claude Code session's state live, shows the aggregate footprint glyph, fires "needs you" notifications, and answers permission prompts (Allow/Deny) in place.

**Architecture:** SPM executable in `app/`. A `BridgeSupervisor` spawns/watches bundled `node server.js`; a `BridgeClient` reads the port file, streams `/events` (SSE) with a `/state` poll fallback, and POSTs `/decision`; a `@MainActor @Observable SessionStore` holds the decoded `Snapshot` and drives the UI + `Notifier`. UI is `MenuBarExtra` → session list, glyph, permission prompt, per `docs/design-brief.md`.

**Tech Stack:** Swift 5.9 (Command Line Tools), SwiftUI, AppKit, `URLSession`, `UserNotifications`, `Foundation.Process`. Tests: an assertion-based executable target (XCTest is unavailable without Xcode).

## Global Constraints

- **No Xcode / no XCTest.** Build with `swift build`. Tests are a separate executable target `ccbar-tests` using `assert`-style checks that exit non-zero on failure. Never use `import XCTest` or `swift test`.
- **Platform floor:** `.macOS(.v13)` (MenuBarExtra requires macOS 13+; dev machine is 26.1).
- **The app is a pure renderer of `/state`** — it never derives session state itself; the bridge is the source of truth.
- **Model matches real captured payloads** (`docs/superpowers/specs/2026-07-11-hook-payloads.md` and the bridge `snapshot()`): `snapshot = { sessions, pending, aggregate, sessionMap, ts }`.
- **Design system:** colors/typography per `docs/design-brief.md` §8 (light+dark). Footprint = session state only; `needs > working > paused > idle`. Color always paired with a glyph/label (color-blind safe).
- **Bundle identity:** `com.shubhankarmohan.footprint`, `LSUIElement = true` (menu-bar only, no Dock icon).
- **Bridge base URL:** `http://127.0.0.1:<port>` where `<port>` is read from `~/.claude-control-bar/port`.

```swift
import Foundation

enum SessionState: String, Codable { case idle, working, needs, paused, ended }
enum Tier: String, Codable { case owned, attached, bestEffort = "best-effort" }
enum Channel: String, Codable { case preToolUse, permissionRequest }

struct Session: Codable, Identifiable, Equatable {
  let id: String
  var cwd: String?
  var tier: Tier?
  var tmux: String?
  var state: SessionState
  var tool: String?
  var lastLine: String?
  var updatedAt: Double?

  var project: String { (cwd?.split(separator: "/").last).map(String.init) ?? "—" }
}

struct Pending: Codable, Identifiable, Equatable {
  let id: String
  let sessionId: String
  var channel: Channel?
  var tool: String?
  var input: [String: JSONValue]?
  var createdAt: Double?

  var command: String? {
    if case let .string(s)? = input?["command"] { return s }
    return nil
  }
}

struct SessionMapEntry: Codable, Equatable {
  var cwd: String?
  var tmux: String?
  var project: String?
}

struct Snapshot: Codable, Equatable {
  var sessions: [Session]
  var pending: [Pending]
  var aggregate: SessionState
  var sessionMap: [String: SessionMapEntry]?
  var ts: Double?

  static let empty = Snapshot(sessions: [], pending: [], aggregate: .idle, sessionMap: [:], ts: nil)

  // Recompute the aggregate locally too (defensive): needs > working > paused > idle.
  static func priority(_ states: [SessionState]) -> SessionState {
    if states.contains(.needs) { return .needs }
    if states.contains(.working) { return .working }
    if states.contains(.paused) { return .paused }
    return .idle
  }
}

// Minimal JSON value so tool_input (arbitrary shape) decodes losslessly.
enum JSONValue: Codable, Equatable {
  case string(String), number(Double), bool(Bool), object([String: JSONValue]), array([JSONValue]), null

  init(from decoder: Decoder) throws {
    let c = try decoder.singleValueContainer()
    if c.decodeNil() { self = .null }
    else if let b = try? c.decode(Bool.self) { self = .bool(b) }
    else if let n = try? c.decode(Double.self) { self = .number(n) }
    else if let s = try? c.decode(String.self) { self = .string(s) }
    else if let o = try? c.decode([String: JSONValue].self) { self = .object(o) }
    else if let a = try? c.decode([JSONValue].self) { self = .array(a) }
    else { self = .null }
  }
  func encode(to encoder: Encoder) throws {
    var c = encoder.singleValueContainer()
    switch self {
    case .string(let s): try c.encode(s)
    case .number(let n): try c.encode(n)
    case .bool(let b): try c.encode(b)
    case .object(let o): try c.encode(o)
    case .array(let a): try c.encode(a)
    case .null: try c.encodeNil()
    }
  }
}
```

enum BridgePaths {
  static var dir: URL {
    FileManager.default.homeDirectoryForCurrentUser.appendingPathComponent(".claude-control-bar")

  static var portFile: URL { dir.appendingPathComponent("port") }

```swift
import Foundation
import Observation

@MainActor
@Observable
final class SessionStore {
  var snapshot: Snapshot = .empty
  var connected: Bool = false
  private var previousNeeds: Set<String> = []

  /// Apply a raw /state or SSE `data:` JSON blob. Returns session ids that newly
  /// entered `needs` (for the Notifier).
  @discardableResult
  func apply(_ raw: Data) -> [String] {
    guard let snap = try? JSONDecoder().decode(Snapshot.self, from: raw) else { return [] }
    let nowNeeds = Set(snap.sessions.filter { $0.state == .needs }.map { $0.id })
    let newly = nowNeeds.subtracting(previousNeeds)
    previousNeeds = nowNeeds
    snapshot = snap
    return Array(newly)
  }
}
```

```swift
import Foundation

/// Streams snapshots from the bridge: SSE when reachable, else short polling.
/// Emits raw `Data` blobs (the store decodes) so decode failures don't kill the stream.
struct BridgeClient {
  func stream() -> AsyncStream<Data> {
    AsyncStream { continuation in
      let task = Task {
        while !Task.isCancelled {
          guard let base = BridgePaths.baseURL() else {
            try? await Task.sleep(nanoseconds: 1_000_000_000); continue
          }
          do {
            var req = URLRequest(url: base.appendingPathComponent("events"))
            req.timeoutInterval = .infinity
            req.setValue("text/event-stream", forHTTPHeaderField: "Accept")
            let (bytes, _) = try await URLSession.shared.bytes(for: req)
            for try await line in bytes.lines {
              if line.hasPrefix("data: ") {
                continuation.yield(Data(line.dropFirst(6).utf8))
              }
            }
          } catch {
            // SSE dropped → poll once, then retry the stream shortly.
            if let base = BridgePaths.baseURL(),
               let (data, _) = try? await URLSession.shared.data(from: base.appendingPathComponent("state")) {
              continuation.yield(data)
            }
            try? await Task.sleep(nanoseconds: 2_000_000_000)
          }
        }
        continuation.finish()
      }
      continuation.onTermination = { _ in task.cancel() }
    }
  }

  func decide(id: String, decision: String, updatedInput: [String: JSONValue]? = nil) async {
    guard let base = BridgePaths.baseURL() else { return }
    var req = URLRequest(url: base.appendingPathComponent("decision"))
    req.httpMethod = "POST"
    req.setValue("application/json", forHTTPHeaderField: "Content-Type")
    var body: [String: Any] = ["id": id, "decision": decision]
    if let ui = updatedInput, let data = try? JSONEncoder().encode(ui),
       let obj = try? JSONSerialization.jsonObject(with: data) { body["updatedInput"] = obj }
    req.httpBody = try? JSONSerialization.data(withJSONObject: body)
    _ = try? await URLSession.shared.data(for: req)
  }
}
```

```swift
import Foundation

/// Spawns and watches the bundled Node bridge. In a built .app the bridge is at
/// Contents/Resources/bridge/server.js; in dev we fall back to the repo path.
final class BridgeSupervisor {
  private var process: Process?

  private func serverJS() -> URL? {
    if let res = Bundle.main.resourceURL {
      let bundled = res.appendingPathComponent("bridge/server.js")
      if FileManager.default.fileExists(atPath: bundled.path) { return bundled }
    }
    // dev fallback: CCBAR_BRIDGE env or a sensible repo-relative guess
    if let env = ProcessInfo.processInfo.environment["CCBAR_BRIDGE"] {
      return URL(fileURLWithPath: env)
    }
    return nil
  }

  private func nodePath() -> String {
    for p in ["/opt/homebrew/bin/node", "/usr/local/bin/node", "/usr/bin/node"] {
      if FileManager.default.fileExists(atPath: p) { return p }
    }
    return "/usr/bin/env" // last resort: resolve via PATH
  }

  func start() {
    guard let js = serverJS() else { return }
    let p = Process()
    let node = nodePath()
    p.executableURL = URL(fileURLWithPath: node)
    p.arguments = node.hasSuffix("env") ? ["node", js.path] : [js.path]
    p.terminationHandler = { [weak self] _ in
      // Restart after a short delay unless we're tearing down.
      DispatchQueue.global().asyncAfter(deadline: .now() + 1) { self?.start() }
    }
    try? p.run()
    process = p
  }

  func stop() {
    process?.terminationHandler = nil
    process?.terminate()
    process = nil
  }
}
```

    switch s {

import SwiftUI

struct PopoverView: View {
  @Bindable var store: SessionStore
  private let client = BridgeClient()

  var body: some View {
    VStack(alignment: .leading, spacing: 0) {
      HStack {
        Text("Claude Control Bar").font(.system(size: 12, weight: .semibold))
        Spacer()
        Circle().fill(store.connected ? .green : .secondary).frame(width: 6, height: 6)
      }.padding(.horizontal, 12).padding(.vertical, 8)
      Divider()

      if store.snapshot.sessions.isEmpty {
        Text("The map is quiet.")
          .font(.system(size: 13)).foregroundStyle(.secondary)
          .frame(maxWidth: .infinity).padding(.vertical, 28)
      } else {
        ForEach(sorted) { s in SessionRowView(session: s) }
      }

      Divider()
      HStack {
        Text("Mischief managed").font(.system(size: 10)).foregroundStyle(.tertiary)
        Spacer()
        Button("Quit") { NSApplication.shared.terminate(nil) }
          .buttonStyle(.plain).font(.system(size: 11)).foregroundStyle(.secondary)
      }.padding(.horizontal, 12).padding(.vertical, 6)
    }
    .frame(width: 320)
    .task { for await raw in client.stream() { store.connected = true; store.apply(raw) } }
  }

  // Priority sort: needs, then working, then others; stable by project.
  private var sorted: [Session] {
    store.snapshot.sessions.sorted { a, b in
      func rank(_ s: SessionState) -> Int { [.needs: 0, .working: 1, .paused: 2, .idle: 3, .ended: 4][s] ?? 5 }
      return rank(a.state) != rank(b.state) ? rank(a.state) < rank(b.state) : a.project < b.project
    }
  }
}

import Foundation
import UserNotifications

enum Notifier {
  static func requestAuth() {
    UNUserNotificationCenter.current().requestAuthorization(options: [.alert, .sound]) { _, _ in }
  }
  static func needsYou(project: String) {
    let c = UNMutableNotificationContent()
    c.title = "Claude needs you"
    c.body = "\(project) is waiting for a permission decision."
    c.sound = .default
    let req = UNNotificationRequest(identifier: UUID().uuidString, content: c, trigger: nil)
    UNUserNotificationCenter.current().add(req)
  }
}
